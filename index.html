    <script>
        const wrapper = document.getElementById('wrapper');
        const slides = document.querySelectorAll('.slide');
        const prevBtn = document.getElementById('prevBtn');
        const nextBtn = document.getElementById('nextBtn');
        const counterDisplay = document.getElementById('counterDisplay');

        let currentIdx = 0;
        let totalSlides = slides.length;

        wrapper.style.width = `${totalSlides * 100}vw`;

        function updateUI() {
            wrapper.style.transform = `translateX(-${currentIdx * 100}vw)`;
            prevBtn.classList.toggle('disabled', currentIdx === 0);
            nextBtn.classList.toggle('disabled', currentIdx === totalSlides - 1);
            counterDisplay.innerText = `0${currentIdx + 1} / 0${totalSlides}`;
        }

        function moveSlide(direction) {
            const nextIdx = currentIdx + direction;
            if (nextIdx >= 0 && nextIdx < totalSlides) {
                currentIdx = nextIdx;
                updateUI();
            }
        }

        nextBtn.onclick = () => moveSlide(1);
        prevBtn.onclick = () => moveSlide(-1);

        let isThrottled = false;
        window.addEventListener('wheel', e => {
            if (isThrottled) return;
            if (Math.abs(e.deltaY) > 30) {
                isThrottled = true;
                moveSlide(e.deltaY > 0 ? 1 : -1);
                setTimeout(() => isThrottled = false, 1200);
            }
        }, { passive: true });

        let touchX = 0;
        window.addEventListener('touchstart', e => touchX = e.touches[0].clientX);
        window.addEventListener('touchend', e => {
            const deltaX = touchX - e.changedTouches[0].clientX;
            if (Math.abs(deltaX) > 50) moveSlide(deltaX > 0 ? 1 : -1);
        });

        updateUI();
    </script>
